# Paid EASM API (docs/PAID_API_DESIGN.md, api/) — a separately deployable
# service, never required by app.py or the CLI. Install with:
#   pip install -r requirements.txt -r requirements-api.txt
#
# FastAPI: confirmed against current docs (fastapi.tiangolo.com) as the
# sensible choice for an async Python service that queues long-running
# work (~25 min scans) without blocking, and serves fast status/report
# GETs — native async def support, actively maintained.
fastapi==0.141.1
# uvicorn[standard]: FastAPI's own recommended ASGI server; the "standard"
# extra pulls in the faster event loop / HTTP parser used for local dev
# and the `--reload` flag.
uvicorn[standard]==0.53.0
# argon2-cffi: API key hashing (docs/PAID_API_DESIGN.md Part C) — Argon2id
# is deliberately slow/salted, unlike the fast SHA256 lookup index stored
# alongside it (see api/security.py).
argon2-cffi==25.1.0
# dnspython: a real DNS TXT query for domain-ownership verification
# (docs/PAID_API_DESIGN.md Part A, api/domain_verification.py::verify_dns_txt)
# — stdlib `socket` has no TXT-record support (only A/AAAA via
# getaddrinfo). Pure-Python, no external `dig`/`nslookup` binary needed,
# so this works the same whether or not the deployment host happens to
# have Hydra's own recon tools (dnsx) installed — the API must not depend
# on those being present.
dnspython==2.8.0
# httpx: now a REAL runtime dependency, not just test tooling — the
# well-known-file verification method
# (api/domain_verification.py::verify_well_known_file) makes a real
# outbound HTTPS GET to the domain being verified. Previously only
# requirements-dev.txt needed this (fastapi.testclient.TestClient); kept
# pinned there too, at the same version, for anyone who installs test
# dependencies without deploying the API. See that file's own comment for
# the console-script name-collision this package's `pip install` causes
# with the ProjectDiscovery `httpx` recon binary — already defended
# against (core/dependencies/), not a new concern introduced here.
httpx==0.28.1
# boto3: optional remote backup upload (docs/PAID_API_DESIGN.md's
# "Automated backups and a real deployment target" section,
# api/backup_worker.py) — S3-compatible, not locked to AWS: any
# provider that speaks the S3 API (DigitalOcean Spaces, Backblaze B2,
# Cloudflare R2) works via `HYDRA_API_BACKUP_S3_ENDPOINT_URL`.
# Installed unconditionally here (small, pure-Python-plus-botocore, no
# C extensions) so the remote-upload code path is genuinely exercised
# in CI rather than skipped — but importing it is still wrapped in a
# runtime check in api/backup_worker.py, since remote upload itself
# stays fully OPT-IN (no bucket configured = local-only backups, never
# a hard failure).
boto3==1.43.100
# sentry-sdk: error tracking (docs/PAID_API_DESIGN.md's "Basic
# observability" section, api/observability.py) — auto-detects the
# installed `fastapi` package and enables its own FastAPI/Starlette
# integration accordingly (confirmed against Sentry's current docs), no
# separate `[fastapi]` extra needed. Entirely opt-in at runtime: unset
# `SENTRY_DSN` means `init_sentry` never calls `sentry_sdk.init()` at
# all.
sentry-sdk==2.70.0
